cmake_minimum_required(VERSION 3.24)
project(batchcrypto VERSION 0.3.0 LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
include(cmake/EmbedP256Tables.cmake)
add_library(batchcrypto SHARED src/abi/batchcrypto_abi.cpp src/engine/crypto_engine.cu src/p256/tables.cpp)
target_include_directories(batchcrypto PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(batchcrypto PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_definitions(batchcrypto PRIVATE BC_BUILD)
set_target_properties(batchcrypto PROPERTIES CXX_STANDARD 17 CUDA_STANDARD 17 POSITION_INDEPENDENT_CODE ON)
target_link_libraries(batchcrypto PRIVATE CUDA::cudart)
include(GNUInstallDirs)
install(TARGETS batchcrypto EXPORT batchcryptoTargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/batchcrypto.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT batchcryptoTargets NAMESPACE batchcrypto:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/batchcrypto)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/batchcryptoConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/batchcryptoConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/batchcrypto)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/batchcryptoConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/batchcryptoConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/batchcryptoConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/batchcrypto)
include(CTest)
if(BUILD_TESTING)
    enable_language(C)
    add_executable(batchcrypto_c_api examples/c_api.c)
    target_link_libraries(batchcrypto_c_api PRIVATE batchcrypto)
    add_test(NAME c_api COMMAND batchcrypto_c_api)
endif()
