cmake_minimum_required(VERSION 3.24)
project(batchcrypto_pipeline LANGUAGES CXX)
find_package(OpenSSL 3.2 REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
add_executable(pipeline pipeline.cpp)
target_compile_features(pipeline PRIVATE cxx_std_17)
target_link_libraries(pipeline PRIVATE OpenSSL::Crypto Threads::Threads)
if(BC_LIBRARY)
  target_compile_definitions(pipeline PRIVATE PIPELINE_GPU=1)
  target_include_directories(pipeline PRIVATE ../../include)
  target_link_libraries(pipeline PRIVATE "${BC_LIBRARY}")
endif()
if(WIN32)
  target_link_libraries(pipeline PRIVATE psapi winmm)
endif()
include(CTest)
add_test(NAME cpu_selftest COMMAND pipeline --selftest)
add_test(NAME cpu_pipeline COMMAND pipeline --mode cpu --rate 1000 --seconds 0.1 --workers 2 --keys 2)
