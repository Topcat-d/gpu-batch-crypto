cmake_minimum_required(VERSION 3.24)
project(batchcrypto_pipeline LANGUAGES CXX)
find_package(OpenSSL 3.2 REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
add_executable(pipeline pipeline.cpp)
add_executable(ready_wave ready_wave.cpp)
foreach(target pipeline ready_wave)
target_compile_features(${target} PRIVATE cxx_std_17)
target_link_libraries(${target} PRIVATE OpenSSL::Crypto Threads::Threads)
if(BC_LIBRARY)
  target_compile_definitions(${target} PRIVATE PIPELINE_GPU=1)
  target_include_directories(${target} PRIVATE ../../include)
  target_link_libraries(${target} PRIVATE "${BC_LIBRARY}")
endif()
if(WIN32)
  target_link_libraries(${target} PRIVATE psapi winmm)
endif()
endforeach()
include(CTest)
add_test(NAME cpu_selftest COMMAND pipeline --selftest)
add_test(NAME cpu_pipeline COMMAND pipeline --mode cpu --rate 1000 --seconds 0.1 --workers 2 --keys 2)
add_test(NAME es256_ready_wave COMMAND ready_wave --mode cpu-es256 --wave 64 --keys 2 --workers 2 --seconds 0.01)
add_test(NAME eddsa_ready_wave COMMAND ready_wave --mode cpu-eddsa --wave 64 --keys 2 --workers 2 --seconds 0.01)
